    <!-- Tema Değiştirme JS -->
    <script>
        let isPink = true;

        function toggleTheme() {
            const body = document.getElementById('site-body');
            const header = document.getElementById('site-header');
            const cardPlayer = document.getElementById('site-card-player');
            const cardComments = document.getElementById('site-card-comments');
            const badge = document.getElementById('site-badge');
            const btnComment = document.getElementById('site-btn-comment');
            const footer = document.getElementById('site-footer');
            const cards = document.querySelectorAll('.site-card');
            const borders = document.querySelectorAll('.theme-border');
            
            const themeIcon = document.getElementById('theme-icon');
            const themeText = document.getElementById('theme-text');

            if (isPink) {
                // Mor Temaya Geç
                body.className = "bg-purpleBg text-purpleText font-sans min-h-screen flex flex-col justify-between transition-colors duration-300";
                header.className = "bg-purpleHeader border-b border-purpleAccent/20 sticky top-0 z-50 shadow-sm transition-colors duration-300";
                cardPlayer.className = "bg-purpleCard p-4 md:p-6 rounded-2xl border border-purpleAccent/20 shadow-md transition-colors duration-300";
                cardComments.className = "bg-purpleCard p-6 rounded-2xl border border-purpleAccent/20 shadow-sm transition-colors duration-300";
                badge.className = "bg-purpleAccent text-white text-xs px-2.5 py-1 rounded-full font-bold uppercase transition-colors duration-300";
                btnComment.className = "mt-2 bg-purpleAccent hover:bg-purpleAccentHover text-white px-5 py-2 rounded-xl text-xs font-bold transition";
                footer.className = "bg-purpleHeader border-t border-purpleAccent/20 py-6 text-center text-sm font-medium transition-colors duration-300";

                cards.forEach(c => c.className = "site-card bg-purpleCard rounded-2xl p-5 border border-purpleAccent/20 shadow-sm transition duration-300");
                borders.forEach(b => b.className = "text-xl font-bold border-l-4 border-purpleAccent pl-3 mb-6 theme-border");

                themeIcon.innerText = "🌸";
                themeText.innerText = "Pembe Tema";
                isPink = false;
            } else {
                // Pembe Temaya Dön
                body.className = "bg-pinkBg text-pinkText font-sans min-h-screen flex flex-col justify-between transition-colors duration-300";
                header.className = "bg-pinkHeader border-b border-pinkAccent/20 sticky top-0 z-50 shadow-sm transition-colors duration-300";
                cardPlayer.className = "bg-pinkCard p-4 md:p-6 rounded-2xl border border-pinkAccent/20 shadow-md transition-colors duration-300";
                cardComments.className = "bg-pinkCard p-6 rounded-2xl border border-pinkAccent/20 shadow-sm transition-colors duration-300";
                badge.className = "bg-pinkAccent text-white text-xs px-2.5 py-1 rounded-full font-bold uppercase transition-colors duration-300";
                btnComment.className = "mt-2 bg-pinkAccent hover:bg-pinkAccentHover text-white px-5 py-2 rounded-xl text-xs font-bold transition";
                footer.className = "bg-pinkHeader border-t border-pinkAccent/20 py-6 text-center text-sm font-medium transition-colors duration-300";

                cards.forEach(c => c.className = "site-card bg-pinkCard rounded-2xl p-5 border border-pinkAccent/20 shadow-sm transition duration-300");
                borders.forEach(b => b.className = "text-xl font-bold border-l-4 border-pinkAccent pl-3 mb-6 theme-border");

                themeIcon.innerText = "🔮";
                themeText.innerText = "Mor Tema";
                isPink = true;
            }
        }
    </script>
